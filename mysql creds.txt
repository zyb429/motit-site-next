-- ============================================
-- schema.sql для motit_site (совместим со Strapi)
--
-- ВАЖНО:
--   * Таблицы users, roles, permissions, posts, categories
--     и все up_*/admin_*/files/strapi_* создаёт Strapi.
--   * Этот скрипт создаёт только бизнес-таблицы.
--   * Запускать ПОСЛЕ первого старта Strapi.
--   * Скрипт идемпотентный: можно запускать повторно.
-- ============================================

CREATE DATABASE IF NOT EXISTS `motit_site`
    DEFAULT CHARACTER SET utf8mb4
    COLLATE utf8mb4_unicode_520_ci;

USE `motit_site`;

-- ============================================
-- 1. organizations
-- ============================================
CREATE TABLE IF NOT EXISTS `organizations` (
    `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    `uuid` CHAR(36) NOT NULL DEFAULT (UUID()),
    `name` VARCHAR(255) NOT NULL,
    `inn` VARCHAR(20) NULL,
    `address` TEXT NULL,
    `email` VARCHAR(255) NULL,
    `phone` VARCHAR(50) NULL,
    `is_active` BOOLEAN DEFAULT TRUE,
    `created_by_id` INT UNSIGNED NULL,
    `updated_by_id` INT UNSIGNED NULL,
    `locale` VARCHAR(255) NULL,
    `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    `updated_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    PRIMARY KEY (`id`),
    UNIQUE KEY `idx_organizations_uuid` (`uuid`),
    KEY `idx_organizations_inn` (`inn`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_520_ci;

-- ============================================
-- 2. clients
-- ============================================
CREATE TABLE IF NOT EXISTS `clients` (
    `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    `uuid` CHAR(36) NOT NULL DEFAULT (UUID()),
    `user_uuid` CHAR(36) NOT NULL,
    `created_by_id` INT UNSIGNED NULL,
    `updated_by_id` INT UNSIGNED NULL,
    `locale` VARCHAR(255) NULL,
    `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    `updated_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    PRIMARY KEY (`id`),
    UNIQUE KEY `idx_clients_uuid` (`uuid`),
    UNIQUE KEY `idx_clients_user_uuid` (`user_uuid`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_520_ci;

-- ============================================
-- 3. client_organizations
-- ============================================
CREATE TABLE IF NOT EXISTS `client_organizations` (
    `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    `client_user_uuid` CHAR(36) NOT NULL,
    `organization_uuid` CHAR(36) NOT NULL,
    `role_in_company` VARCHAR(50) DEFAULT 'member',
    `is_primary` BOOLEAN DEFAULT FALSE,
    `created_by_id` INT UNSIGNED NULL,
    `updated_by_id` INT UNSIGNED NULL,
    `locale` VARCHAR(255) NULL,
    `joined_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    `updated_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    PRIMARY KEY (`id`),
    UNIQUE KEY `idx_client_org_unique` (`client_user_uuid`, `organization_uuid`),
    KEY `idx_client_org_client` (`client_user_uuid`),
    KEY `idx_client_org_org` (`organization_uuid`),
    CONSTRAINT `fk_client_org_org`
      FOREIGN KEY (`organization_uuid`) REFERENCES `organizations` (`uuid`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_520_ci;

-- ============================================
-- 4. workers
-- ============================================
CREATE TABLE IF NOT EXISTS `workers` (
    `user_uuid` CHAR(36) NOT NULL,
    `position` VARCHAR(255) NULL,
    `created_by_id` INT UNSIGNED NULL,
    `updated_by_id` INT UNSIGNED NULL,
    `locale` VARCHAR(255) NULL,
    `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    `updated_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    PRIMARY KEY (`user_uuid`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_520_ci;

-- ============================================
-- 5. statuses
-- ============================================
CREATE TABLE IF NOT EXISTS `statuses` (
    `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    `uuid` CHAR(36) NOT NULL DEFAULT (UUID()),
    `name` VARCHAR(100) NOT NULL,
    `created_by_id` INT UNSIGNED NULL,
    `updated_by_id` INT UNSIGNED NULL,
    `locale` VARCHAR(255) NULL,
    `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    `updated_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    PRIMARY KEY (`id`),
    UNIQUE KEY `idx_statuses_uuid` (`uuid`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_520_ci;

-- ============================================
-- 6. priorities
-- ============================================
CREATE TABLE IF NOT EXISTS `priorities` (
    `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    `uuid` CHAR(36) NOT NULL DEFAULT (UUID()),
    `name` VARCHAR(100) NOT NULL,
    `created_by_id` INT UNSIGNED NULL,
    `updated_by_id` INT UNSIGNED NULL,
    `locale` VARCHAR(255) NULL,
    `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    `updated_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    PRIMARY KEY (`id`),
    UNIQUE KEY `idx_priorities_uuid` (`uuid`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_520_ci;

-- ============================================
-- 7. tickets
-- ============================================
CREATE TABLE IF NOT EXISTS `tickets` (
    `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    `uuid` CHAR(36) NOT NULL DEFAULT (UUID()),
    `title` VARCHAR(255) NOT NULL,
    `description` TEXT NULL,
    `client_uuid` CHAR(36) NULL,
    `assigned_to_uuid` CHAR(36) NULL,
    `status_uuid` CHAR(36) NULL,
    `priority_uuid` CHAR(36) NULL,
    `category_uuid` CHAR(36) NULL,
    `deadline_at` TIMESTAMP NULL,
    `resolved_at` TIMESTAMP NULL,
    `created_by_id` INT UNSIGNED NULL,
    `updated_by_id` INT UNSIGNED NULL,
    `locale` VARCHAR(255) NULL,
    `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    `updated_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    PRIMARY KEY (`id`),
    UNIQUE KEY `idx_tickets_uuid` (`uuid`),
    KEY `idx_tickets_client` (`client_uuid`),
    KEY `idx_tickets_assigned` (`assigned_to_uuid`),
    KEY `idx_tickets_status` (`status_uuid`),
    KEY `idx_tickets_priority` (`priority_uuid`),
    KEY `idx_tickets_category` (`category_uuid`),
    KEY `idx_tickets_created_at` (`created_at`),
    KEY `idx_tickets_deadline_at` (`deadline_at`),
    CONSTRAINT `fk_tickets_status`
      FOREIGN KEY (`status_uuid`) REFERENCES `statuses` (`uuid`) ON DELETE SET NULL,
    CONSTRAINT `fk_tickets_priority`
      FOREIGN KEY (`priority_uuid`) REFERENCES `priorities` (`uuid`) ON DELETE SET NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_520_ci;

-- ============================================
-- 8. ticket_comments
-- ============================================
CREATE TABLE IF NOT EXISTS `ticket_comments` (
    `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    `uuid` CHAR(36) NOT NULL DEFAULT (UUID()),
    `ticket_uuid` CHAR(36) NOT NULL,
    `user_uuid` CHAR(36) NOT NULL,
    `content` TEXT NOT NULL,
    `is_internal` BOOLEAN DEFAULT FALSE,
    `created_by_id` INT UNSIGNED NULL,
    `updated_by_id` INT UNSIGNED NULL,
    `locale` VARCHAR(255) NULL,
    `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    `updated_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    PRIMARY KEY (`id`),
    UNIQUE KEY `idx_ticket_comments_uuid` (`uuid`),
    KEY `idx_ticket_comments_ticket` (`ticket_uuid`),
    KEY `idx_ticket_comments_user` (`user_uuid`),
    CONSTRAINT `fk_ticket_comments_ticket`
      FOREIGN KEY (`ticket_uuid`) REFERENCES `tickets` (`uuid`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_520_ci;

-- ============================================
-- 9. pages
-- ============================================
CREATE TABLE IF NOT EXISTS `pages` (
    `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    `uuid` CHAR(36) NOT NULL DEFAULT (UUID()),
    `slug` VARCHAR(255) NOT NULL,
    `title` VARCHAR(255) NOT NULL,
    `sections` JSON NULL,
    `seo_data` JSON NULL,
    `meta_title` VARCHAR(255) NULL,
    `meta_description` VARCHAR(300) NULL,
    `created_by_id` INT UNSIGNED NULL,
    `updated_by_id` INT UNSIGNED NULL,
    `locale` VARCHAR(255) NULL,
    `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    `updated_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    PRIMARY KEY (`id`),
    UNIQUE KEY `idx_pages_uuid` (`uuid`),
    UNIQUE KEY `idx_pages_slug` (`slug`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_520_ci;

-- ============================================
-- 10. issuers
-- ============================================
CREATE TABLE IF NOT EXISTS `issuers` (
    `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    `uuid` CHAR(36) NOT NULL DEFAULT (UUID()),
    `name` VARCHAR(255) NOT NULL,
    `website_url` VARCHAR(255) NULL,
    `created_by_id` INT UNSIGNED NULL,
    `updated_by_id` INT UNSIGNED NULL,
    `locale` VARCHAR(255) NULL,
    `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    `updated_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    PRIMARY KEY (`id`),
    UNIQUE KEY `idx_issuers_uuid` (`uuid`),
    UNIQUE KEY `idx_issuers_name` (`name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_520_ci;

-- ============================================
-- 11. certificates
-- ============================================
CREATE TABLE IF NOT EXISTS `certificates` (
    `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    `uuid` CHAR(36) NOT NULL DEFAULT (UUID()),
    `title` VARCHAR(255) NOT NULL,
    `issuer_id` BIGINT UNSIGNED NULL,
    `image_url` VARCHAR(500) NULL,
    `issue_date` DATE NULL,
    `expiry_date` DATE NULL,
    `is_active` BOOLEAN DEFAULT TRUE,
    `order` INT DEFAULT 0,
    `created_by_id` INT UNSIGNED NULL,
    `updated_by_id` INT UNSIGNED NULL,
    `locale` VARCHAR(255) NULL,
    `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    `updated_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    PRIMARY KEY (`id`),
    UNIQUE KEY `idx_certificates_uuid` (`uuid`),
    KEY `idx_certificates_issuer` (`issuer_id`),
    CONSTRAINT `fk_certificates_issuer`
      FOREIGN KEY (`issuer_id`) REFERENCES `issuers` (`id`) ON DELETE SET NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_520_ci;

-- ============================================
-- 12. partners
-- ============================================
CREATE TABLE IF NOT EXISTS `partners` (
    `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    `uuid` CHAR(36) NOT NULL DEFAULT (UUID()),
    `name` VARCHAR(255) NOT NULL,
    `website_url` VARCHAR(255) NULL,
    `image_url` VARCHAR(500) NULL,
    `order` INT DEFAULT 0,
    `created_by_id` INT UNSIGNED NULL,
    `updated_by_id` INT UNSIGNED NULL,
    `locale` VARCHAR(255) NULL,
    `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    `updated_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    PRIMARY KEY (`id`),
    UNIQUE KEY `idx_partners_uuid` (`uuid`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_520_ci;
