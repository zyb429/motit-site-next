-- ============================================
-- 0. Удаление старой БД (если существует)
-- ============================================
DROP DATABASE IF EXISTS `motit_site`;

-- ============================================
-- 1. Создание БД
-- ============================================
CREATE DATABASE `motit_site` 
    DEFAULT CHARACTER SET utf8mb4 
    COLLATE utf8mb4_unicode_520_ci;

USE `motit_site`;

-- ============================================
-- 2. Таблица: users (исправленная для Strapi)
-- ============================================
CREATE TABLE `users` (
    `id` INT UNSIGNED NOT NULL AUTO_INCREMENT,
    `uuid` CHAR(36) NOT NULL DEFAULT (UUID()),
    `document_id` VARCHAR(255) NULL,
    `username` VARCHAR(255) NULL,
    `email` VARCHAR(255) NULL,
    `password` VARCHAR(255) NULL,
    `password_hash` VARCHAR(255) NULL,
    `full_name` VARCHAR(255) NULL,
    `phone` VARCHAR(255) NULL,
    `avatar_url` VARCHAR(500) NULL,
    `is_active` TINYINT(1) DEFAULT 1,
    `last_login` DATETIME(6) NULL,
    `provider` VARCHAR(255) NULL,
    `confirmed` TINYINT(1) DEFAULT 0,
    `blocked` TINYINT(1) DEFAULT 0,
    `resetPasswordToken` VARCHAR(255) NULL,
    `confirmationToken` VARCHAR(255) NULL,
    `created_at` DATETIME(6) NULL,
    `updated_at` DATETIME(6) NULL,
    `published_at` DATETIME(6) NULL,
    `created_by_id` INT UNSIGNED NULL,
    `updated_by_id` INT UNSIGNED NULL,
    `locale` VARCHAR(255) NULL,
    PRIMARY KEY (`id`),
    UNIQUE KEY `idx_users_uuid` (`uuid`),
    UNIQUE KEY `idx_users_username` (`username`),
    UNIQUE KEY `idx_users_email` (`email`),
    KEY `idx_users_document_id` (`document_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_520_ci;

-- ============================================
-- 3. Таблица: roles (исправленная)
-- ============================================
CREATE TABLE `roles` (
    `id` INT UNSIGNED NOT NULL AUTO_INCREMENT,  -- ← INT, не BIGINT
    `uuid` CHAR(36) NOT NULL DEFAULT (UUID()),
    `name` VARCHAR(100) NOT NULL,
    `created_by_id` INT UNSIGNED NULL,
    `updated_by_id` INT UNSIGNED NULL,
    `locale` VARCHAR(255) NULL,
    `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    `updated_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    PRIMARY KEY (`id`),
    UNIQUE KEY `idx_roles_uuid` (`uuid`),
    UNIQUE KEY `idx_roles_name` (`name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_520_ci;

-- ============================================
-- 3b. Таблица: users_role_lnk (ИСПРАВЛЕНА)
-- ============================================
CREATE TABLE `users_role_lnk` (
    `id` INT UNSIGNED NOT NULL AUTO_INCREMENT,
    `user_id` INT UNSIGNED NOT NULL,
    `role_id` INT UNSIGNED NOT NULL,
    `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    `updated_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    PRIMARY KEY (`id`),
    UNIQUE KEY `idx_users_role_lnk_unique` (`user_id`, `role_id`),
    FOREIGN KEY (`user_id`) REFERENCES `users`(`id`) ON DELETE CASCADE,
    FOREIGN KEY (`role_id`) REFERENCES `roles`(`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_520_ci;

    `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
